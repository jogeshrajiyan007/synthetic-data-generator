CREATE OR REPLACE TEMPORARY VIEW MOLCustomerAgentResponse AS
WITH ordered AS (
  SELECT
    *,
    LAG(MessageCreatedBy2)   OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevSender,
    LAG(SecureMessageId)     OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevSecureMessageId,
    LAG(MessageTimestamp)    OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevMessageTimestamp,
    LAG(AH_MessageContent_c) OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevMessageContent
  FROM MOLClean
  WHERE MessageCreatedBy2 = 'Agent'
)
SELECT
  ord.CaseNumber,
  ord.policynumber,
  ord.SecureMessageId    AS AgentSecureMessageId,
  ord.MessageTimestamp   AS AgentMessageTimestamp,
  ord.AH_MessageContent_c AS AgentMessage,
  -- Response metadata from MOLCustomerMessage (null for Customer_Initiation rows)
  COALESCE(c.ResponseAgentId, ord.ResponseAgentId) AS ResponseAgentId,
  COALESCE(c.AgentName, ord.AgentName) AS AgentName,
  COALESCE(c.HasTransferFlag, ord.HasTransferFlag) AS HasTransferFlag,
  COALESCE(c.HandleTimeSecs, ord.HandleTimeSecs) AS HandleTimeSecs,
  COALESCE(c.wait_time, ord.wait_time) AS wait_time,
  COALESCE(c.ContactChannel, ord.ContactChannel) AS ContactChannel,
  -- Customer message context from LAG (if exists)
  ord.PrevSecureMessageId  AS CustomerSecureMessageId,
  ord.PrevMessageTimestamp AS CustomerMessageTimestamp,
  CASE
    WHEN ord.PrevSender = 'Customer' THEN ord.PrevMessageContent
    ELSE '[NO_EXPLICIT_CUSTOMER_MESSAGE]'
  END AS CustomerMessage,
  -- Message type classification
  CASE
    WHEN ord.PrevSender = 'Customer' THEN 'Customer_Query'
    WHEN ord.PrevSender = 'Agent'    THEN 'Agent_FollowUp'
    ELSE 'Customer_Initiation'
  END AS MessageInteractionType,
  -- Follow-up depth counter
  SUM(CASE 
    WHEN ord.PrevSender = 'Customer' OR ord.PrevSender IS NULL THEN 0 
    ELSE 1 
  END)
    OVER (PARTITION BY ord.CaseNumber ORDER BY ord.MessageTimestamp, ord.MessageSequenceNumberInCase
          ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS FollowUpDepth
FROM ordered ord
LEFT JOIN MOLCustomerMessage c
  ON ord.CaseNumber = c.CaseNumber
  AND ord.MessageTimestamp = c.ResponseTimestamp
ORDER BY ord.CaseNumber, ord.AgentMessageTimestamp;
