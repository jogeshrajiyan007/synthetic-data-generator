base = ["correlation_id","sec_ord","stmt_idx","clause_idx","text","norm_key",
        "DetectedRule","MessageType","Confidence"]
whole = (stmts.filter("section IN ('REASON','NEXT')")
  .select("correlation_id","sec_ord","stmt_idx", F.lit(0).alias("clause_idx"),
          F.col("Statement").alias("text"), "norm_key", "DetectedRule", "MessageType",
          F.lit("HIGH").alias("Confidence")))
acts = classified.select("correlation_id","sec_ord","stmt_idx","clause_idx",
          F.col("clause_text").alias("text"), F.col("NormalizedStatement").alias("norm_key"),
          "DetectedRule","MessageType","Confidence")

w = Window.partitionBy("correlation_id","MessageType","norm_key") \
          .orderBy("sec_ord","stmt_idx","clause_idx")
audit = whole.select(base).unionByName(acts.select(base)) \
             .withColumn("is_duplicate", F.row_number().over(w) > 1)

def joined(mt):
    items = F.collect_list(F.when((c("MessageType") == mt) & ~c("is_duplicate"),
                F.struct("sec_ord","stmt_idx","clause_idx","text")))
    return F.concat_ws("\n", F.transform(F.sort_array(items), lambda x: x["text"]))

agg = audit.groupBy("correlation_id").agg(
    joined("CUSTOMER").alias("CustomerMessage"), joined("AGENT").alias("AgentMessage"),
    F.sum((c("Confidence") == "LOW").cast("int")).alias("low_conf_count"))

final = (s2.select("correlation_id","ReasonForCall","ActionsAndAgreements","NextSteps","parse_status")
         .join(agg, "correlation_id", "left")
         .fillna({"CustomerMessage": "", "AgentMessage": "", "low_conf_count": 0}))
