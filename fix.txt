CREATE OR REPLACE TEMPORARY VIEW MOLCustomerAgentResponse AS
SELECT
    COALESCE(c.casenumber, a.CaseNumber)       AS CaseNumber,
    COALESCE(c.policynumber, a.policynumber)   AS policynumber,
    c.SecureMessageId,
    c.MessageTimestamp                         AS CustomerMessageTimestamp,
    c.MessageSequenceNumberInCase,
    c.CustomerMessageSequenceNumberInCase,
    COALESCE(c.AH_MessageContent_c, 'CUSTOMER_INITIATED_MOL_NO_EXPLICIT_MESSAGE') AS CustomerMessage,
    c.ResponseTimestamp,
    a.SecureMessageId                          AS AgentSecureMessageId,
    a.MessageTimestamp                         AS AgentMessageTimestamp,
    a.AgentMessageSequenceNumberInCase,
    a.AH_MessageContent_c                      AS AgentMessage,
    COALESCE(c.ResponseAgentId, a.ResponseAgentId) AS ResponseAgentId,
    COALESCE(c.AgentName, a.AgentName)             AS AgentName,
    COALESCE(c.HasTransferFlag, a.HasTransferFlag) AS HasTransferFlag,
    COALESCE(c.HandleTimeSecs, a.HandleTimeSecs)   AS HandleTimeSecs,
    COALESCE(c.wait_time, a.wait_time)             AS wait_time,
    COALESCE(c.ContactChannel, a.ContactChannel)   AS ContactChannel,
    c.PreviousMessageTimestamp,
    c.GapFromPreviousMessageSecs,
    CASE WHEN c.SecureMessageId IS NULL THEN 'Customer_Initiation' ELSE 'Customer_Query' END AS CustomerInitiationType
FROM MOLCustomerMessage c
RIGHT JOIN MOLClean a
  ON c.casenumber = a.CaseNumber
 AND a.MessageTimestamp = c.ResponseTimestamp
WHERE a.MessageCreatedBy2 = 'Agent'
;
