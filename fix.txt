import json
import time

# Small synthetic transcript for testing
test_transcript = """
Telephone Call Start: 2026-09-21T10:30:00

Customer: I have already submitted a claim for my shoulder injury last week.
Customer: I just want to know whether the claim has been received and what
happens next.

Agent: Yes, I can confirm that we have received your claim.
Agent: It is currently being reviewed and you should receive an update shortly.

Customer: Okay, thank you. That answers my question.
"""

print("Testing call_model()...")
print("-" * 60)

start = time.time()

try:
    # Build the same prompt used by the production workflow
    prompt = build_prompt(test_transcript)

    # Directly test call_model()
    result, meta = call_model(
        prompt,
        max_tokens=800,
        retries=3
    )

    elapsed = time.time() - start

    print("✅ call_model() SUCCESS")
    print(f"Elapsed time:      {elapsed:.2f}s")
    print(f"API latency:       {meta.get('api_latency', 0):.2f}s")
    print(f"Attempt:           {meta.get('attempt')}")
    print(f"Request ID:        {meta.get('request_id')}")
    print(f"Prompt tokens:     {meta.get('prompt_tokens')}")
    print(f"Completion tokens: {meta.get('completion_tokens')}")
    print(f"Total tokens:      {meta.get('total_tokens')}")
    
    print("\n--- MODEL RESPONSE ---")
    print(json.dumps(result, indent=2, ensure_ascii=False))

    # Basic sanity checks
    assert isinstance(result, dict), "Model response is not a dictionary"
    assert "topics" in result, "Missing 'topics' in response"
    assert "overall_resolution" in result, "Missing 'overall_resolution'"

    print("\n✅ Response structure looks valid.")

except Exception as e:
    elapsed = time.time() - start

    print("❌ call_model() FAILED")
    print(f"Elapsed time: {elapsed:.2f}s")
    print(f"Error type: {type(e).__name__}")
    print(f"Error: {e}")

    import traceback
    traceback.print_exc()
