```sql

-- =====================================================================
-- CREATE CHECKPOINT TABLE
-- =====================================================================
-- This table stores intermediate results as LLM processes rows
-- Allows resumption if job is interrupted

CREATE TABLE IF NOT EXISTS axahealth_dataplatform_pd_lab.axahealth_data_scientist_analytics.fcr_telephony_llm_messages_checkpoint (
    ConversationId STRING,
    ClaimNumber STRING,
    ConversationStartTimestamp TIMESTAMP,
    MembershipNumber STRING,
    CustomerMessage STRING,
    AgentMessage STRING,
    Confidence DOUBLE,
    prompt_tokens INT,
    completion_tokens INT,
    total_tokens INT,
    api_latency DOUBLE,
    task_latency DOUBLE,
    request_id STRING,
    error STRING,
    processed_at TIMESTAMP
)
USING DELTA
COMMENT "Checkpoint for FCR telephony LLM message extraction batch processing";

-- =====================================================================
-- CREATE OUTPUT TABLE
-- =====================================================================
-- Final table with all extracted messages after batch completes

CREATE TABLE IF NOT EXISTS axahealth_dataplatform_pd_lab.axahealth_data_scientist_analytics.fcr_telephony_llm_messages_output (
    ConversationId STRING,
    ClaimNumber STRING,
    ConversationStartTimestamp TIMESTAMP,
    MembershipNumber STRING,
    CustomerMessage STRING,
    AgentMessage STRING,
    Confidence DOUBLE,
    prompt_tokens INT,
    completion_tokens INT,
    total_tokens INT,
    api_latency DOUBLE,
    task_latency DOUBLE,
    request_id STRING,
    error STRING,
    processed_at TIMESTAMP
)
USING DELTA
COMMENT "Final output table with customer/agent messages extracted from FCR telephony transcripts";

-- Verify tables created
SHOW TABLES LIKE 'fcr_telephony_llm*';
```

```python
total_records = 0

for start_date, end_date in date_chunks:
    try:
        df_chunk = df_sp.filter(
            (F.col("ConversationStartTimestamp") >= start_date) &
            (F.col("ConversationStartTimestamp") <= end_date)
        )
        count = df_chunk.count()
        print(f"[CHUNK] {start_date} to {end_date}: {count:,} records")

            pandas_dfs.append(df_chunk.toPandas())

        print(f"[WARN] Could not load chunk {start_date}-{end_date}: {e}")
```

## Part 10: Join Back to fcr_telephony_base

```sql
%sql

-- =====================================================================
-- JOIN LLM RESULTS BACK TO fcr_telephony_base
-- =====================================================================

CREATE OR REPLACE TEMPORARY TABLE fcr_telephony_with_messages AS
SELECT 
  a.*,
  b.CustomerMessage,
  b.AgentMessage,
  b.Confidence as MessageExtractionConfidence,
  b.error as MessageExtractionError,
  b.prompt_tokens,
  b.completion_tokens,
  b.total_tokens,
  b.api_latency
FROM fcr_telephony_base a
LEFT JOIN axahealth_dataplatform_pd_lab.axahealth_data_scientist_analytics.fcr_telephony_llm_messages_output b
  ON a.ConversationId = b.ConversationId;

-- Verify join
SELECT COUNT(*) as total,
       COUNT(CASE WHEN CustomerMessage IS NOT NULL THEN 1 END) as has_customer_msg,
       COUNT(CASE WHEN AgentMessage IS NOT NULL THEN 1 END) as has_agent_msg,
       COUNT(CASE WHEN MessageExtractionError IS NULL THEN 1 END) as successful
FROM fcr_telephony_with_messages;

-- Preview
SELECT ConversationId, ClaimNumber, Confidence, CustomerMessage, AgentMessage, MessageExtractionError
FROM fcr_telephony_with_messages
WHERE MessageExtractionError IS NULL AND Confidence > 0.7
LIMIT 5;
```

### If token limit exceeded
```python

MAX_CONCURRENCY = 10  # from 15
SAFETY_BUFFER = 0.85  # from 0.90
```

### If resuming after failure
```python
# The checkpoint table will automatically skip already-processed rows
# Just run the batch processor again with same parameters
```

### If transcripts are too short after PII redaction
