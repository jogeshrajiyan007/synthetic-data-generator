test_sdf = (
    mol_sdf
    .select(
        "casenumber",
        "CustomerEpisodeConversation",
        "AgentEpisodeConversation"
    )
    .where(
        F.col("CustomerEpisodeConversation").isNotNull()
    )
    .limit(10)
)

test_df = test_sdf.toPandas()

test_results = []

for _, row in test_df.iterrows():

    transcript = build_mol_transcript(
        row["CustomerEpisodeConversation"],
        row["AgentEpisodeConversation"]
    )

    prompt = build_mol_prompt(transcript)

    (
        analysis,
        prompt_tokens,
        completion_tokens,
        total_tokens,
        latency,
        error
    ) = call_mol_model(prompt)

    test_results.append({

        "casenumber": row["casenumber"],

        "analysis": analysis,

        "prompt_tokens": prompt_tokens,

        "completion_tokens": completion_tokens,

        "total_tokens": total_tokens,

        "latency": latency,

        "error": error
    })


test_results_df = pd.DataFrame(test_results)

display(test_results_df)
