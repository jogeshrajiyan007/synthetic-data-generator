def _analyse_rule_based(text):
    o = {
        "subject_text": None, "subject_role": "NONE", "voice": "ACTIVE", "by_role": "NONE",
        "root_text": None, "root_lemma": None, "state_pred": False, "role_to_inf": False,
        "has_modal": False, "conj_voice_mismatch": False, "parse_ok": False
    }
    
    if not text or len(text.split()) < 2:
        return o
    
    words = text.split()
    
    verb_idx = None
    root_verb = None
    for i, w in enumerate(words):
        wl = w.lower().rstrip(".,;:!?")
        if wl in PASSIVE_VERBS or wl.endswith("ed") or wl in MODALS:
            verb_idx = i
            root_verb = wl
            break
    
    if verb_idx is None:
        return o
    
    o["has_modal"] = any(w.lower() in MODALS for w in words[:verb_idx+1])
    
    subject_words = []
    subject_role = "NONE"
    for i in range(verb_idx):
        r = role_of(words[i])
        if r != "OTHER":
            subject_words = words[i:i+1]
            subject_role = r
            break
    
    if subject_role == "NONE" and verb_idx > 0:
        first = words[0].lower()
        if first in ("he","she","it","they","we","you"): 
            subject_role = "OTHER"
            subject_words = [words[0]]
    
    if subject_words:
        o["subject_text"] = " ".join(subject_words)
        o["subject_role"] = subject_role
    
    # ---- CRITICAL FIX: Detect implicit passives ----
    # If subject is CUSTOMER and verb is passive-meaning (informed, advised, provided, assured, etc.)
    # and there's no direct object (no role noun immediately after verb), treat as PASSIVE
    implicit_passive = False
    if subject_role == "CUSTOMER" and root_verb in ("informed", "advised", "provided", "assured", 
                                                     "told", "notified", "updated", "given"):
        # Check for direct object: next word after verb should not be a role noun or action target
        if verb_idx + 1 < len(words):
            next_after_verb = words[verb_idx + 1].lower()
            # If next word is "that", "of", "they", "no", etc., it's prep/conj, not a direct object
            if next_after_verb in ("that", "of", "they", "no", "the", "with", "if", "to", "it"):
                implicit_passive = True
    
    # ---- Voice detection (explicit + implicit) ----
    if verb_idx > 0 and words[verb_idx-1].lower() in ("was","were"):
        o["voice"] = "PASSIVE"
        o["root_text"] = words[verb_idx]
        o["root_lemma"] = LEMMAS.get(root_verb, root_verb)
    elif implicit_passive:
        o["voice"] = "PASSIVE"
        o["root_text"] = root_verb
        o["root_lemma"] = LEMMAS.get(root_verb, root_verb)
    elif root_verb == "be" and verb_idx + 1 < len(words):
        o["voice"] = "COPULA"
        o["root_text"] = "be"
        o["root_lemma"] = "be"
        o["state_pred"] = words[verb_idx + 1].lower() in CUSTOMER_STATES
    else:
        o["voice"] = "ACTIVE"
        o["root_text"] = root_verb
        o["root_lemma"] = LEMMAS.get(root_verb, root_verb)
    
    # by-phrase detection
    by_idx = None
    for i, w in enumerate(words):
        if w.lower() == "by" and i + 1 < len(words):
            by_idx = i + 1
            break
    
    if by_idx:
        by_word = words[by_idx]
        o["by_role"] = role_of(by_word)
    
    # Role to infinitive: "Caller to contact..."
    if verb_idx + 1 < len(words) and words[verb_idx + 1].lower() == "to":
        o["role_to_inf"] = subject_role in ("CUSTOMER", "AGENT", "THIRD_PARTY")
    
    # State predicate for copula
    if o["voice"] == "COPULA" and verb_idx + 1 < len(words):
        next_word = words[verb_idx + 1].lower().rstrip(".,;:!?")
        o["state_pred"] = next_word in CUSTOMER_STATES
    
    o["parse_ok"] = True
    return o
