CREATE OR REPLACE TEMPORARY VIEW MOLCustomerAgentResponse AS
WITH ordered AS (
  SELECT
    *,
    LAG(MessageCreatedBy2)   OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevSender,
    LAG(SecureMessageId)     OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevSecureMessageId,
    LAG(MessageTimestamp)    OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevMessageTimestamp,
    LAG(AH_MessageContent_c) OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase) AS PrevMessageContent
  FROM MOLClean
)
SELECT
  CaseNumber,
  policynumber,
  SecureMessageId    AS AgentSecureMessageId,
  MessageTimestamp   AS AgentMessageTimestamp,
  AH_MessageContent_c AS AgentMessage,
  ResponseAgentId,
  AgentName,
  HasTransferFlag,
  HandleTimeSecs,
  wait_time,
  ContactChannel,
  PrevSecureMessageId  AS CustomerSecureMessageId,
  PrevMessageTimestamp AS CustomerMessageTimestamp,
  CASE
    WHEN PrevSender = 'Customer' THEN PrevMessageContent
    ELSE '[NO_EXPLICIT_CUSTOMER_MESSAGE]'
  END AS CustomerMessage,
  CASE
    WHEN PrevSender = 'Customer' THEN 'Customer_Query'
    WHEN PrevSender = 'Agent'    THEN 'Agent_FollowUp'
    ELSE 'Customer_Initiation'
  END AS MessageInteractionType,
  -- how many agent messages have stacked since the last real customer message
  SUM(CASE WHEN PrevSender = 'Customer' OR PrevSender IS NULL THEN 0 ELSE 1 END)
      OVER (PARTITION BY CaseNumber ORDER BY MessageTimestamp, MessageSequenceNumberInCase
            ROWS BETWEEN UNBOUNDED PRECEDING AND CURRENT ROW) AS FollowUpDepth
FROM ordered
WHERE MessageCreatedBy2 = 'Agent'
ORDER BY CaseNumber, AgentMessageTimestamp;
