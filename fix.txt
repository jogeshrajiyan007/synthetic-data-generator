# ---- Stage 6: rules ----
def add_rules(df):
    c = F.col
    dec = (F.when(c("role_to_inf"), F.struct(F.lit("ROLE_TO_INFINITIVE").alias("DetectedRule"), 
                                             F.lit("AGENT").alias("MessageType"), 
                                             F.lit("LOW").alias("Confidence")))
     .when(c("subject_role") == "NONE",    F.struct(F.lit("NO_SUBJECT").alias("DetectedRule"), 
                                                     F.lit("AGENT").alias("MessageType"), 
                                                     F.lit("LOW").alias("Confidence")))
     .when(c("subject_role") == "PRONOUN", F.struct(F.lit("UNRESOLVED_PRONOUN").alias("DetectedRule"), 
                                                     F.lit("AGENT").alias("MessageType"), 
                                                     F.lit("LOW").alias("Confidence")))
     .when(c("subject_role") == "BOTH",    F.struct(F.lit("JOINT_SUBJECT").alias("DetectedRule"), 
                                                     F.lit("AGENT").alias("MessageType"), 
                                                     F.lit("LOW").alias("Confidence")))
     .when((c("voice") == "PASSIVE") & (c("by_role") == "CUSTOMER"), 
           F.struct(F.lit("PASSIVE_BY_CUSTOMER").alias("DetectedRule"), 
                    F.lit("CUSTOMER").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .when((c("voice") == "PASSIVE") & c("by_role").isin("AGENT","THIRD_PARTY","OTHER"), 
           F.struct(F.lit("PASSIVE_BY_OTHER").alias("DetectedRule"), 
                    F.lit("AGENT").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE") & c("state_pred"), 
           F.struct(F.lit("CUSTOMER_STATE").alias("DetectedRule"), 
                    F.lit("CUSTOMER").alias("MessageType"), 
                    F.lit("MEDIUM").alias("Confidence")))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE"), 
           F.struct(F.lit("CUSTOMER_AS_RECIPIENT").alias("DetectedRule"), 
                    F.lit("AGENT").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "COPULA") & c("state_pred"), 
           F.struct(F.lit("CUSTOMER_STATE").alias("DetectedRule"), 
                    F.lit("CUSTOMER").alias("MessageType"), 
                    F.lit("MEDIUM").alias("Confidence")))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "COPULA"), 
           F.struct(F.lit("COPULA_UNKNOWN_PREDICATE").alias("DetectedRule"), 
                    F.lit("AGENT").alias("MessageType"), 
                    F.lit("LOW").alias("Confidence")))
     .when(c("subject_role") == "CUSTOMER", 
           F.struct(F.lit("CUSTOMER_AS_ACTOR").alias("DetectedRule"), 
                    F.lit("CUSTOMER").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .when(c("subject_role") == "AGENT",    
           F.struct(F.lit("AGENT_ACTOR").alias("DetectedRule"), 
                    F.lit("AGENT").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .when(c("subject_role") == "THIRD_PARTY", 
           F.struct(F.lit("THIRD_PARTY_ACTOR").alias("DetectedRule"), 
                    F.lit("AGENT").alias("MessageType"), 
                    F.lit("HIGH").alias("Confidence")))
     .otherwise(F.struct(F.lit("OTHER_ACTOR").alias("DetectedRule"), 
                         F.lit("AGENT").alias("MessageType"), 
                         F.lit("MEDIUM").alias("Confidence"))))
    
    return (df.withColumn("dec", dec).select("*", "dec.*").drop("dec")
      .withColumn("Confidence", F.when(c("conj_voice_mismatch") & (c("Confidence") == "HIGH"), "LOW")
                                 .otherwise(c("Confidence")))
      .withColumn("DetectedActor", c("subject_text"))
      .withColumn("IsCustomerSubject", c("subject_role") == "CUSTOMER")
      .withColumn("IsCustomerPassive", (c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE"))
      .withColumn("NormalizedStatement", norm_key("clause_text")))

print("Stage 6: applying rules...")
classified_with_rules = add_rules(classified)
classified_with_rules.select("correlation_id","clause_text","DetectedRule","MessageType","Confidence").limit(10).display()

# ---- Stage 7: dedup and aggregate ----
base = ["correlation_id","sec_ord","stmt_idx","clause_idx","text","norm_key",
        "DetectedRule","MessageType","Confidence"]

# Statements from REASON and NEXT (already classified)
whole = (stmts.filter("section IN ('REASON','NEXT')")
  .select("correlation_id","sec_ord","stmt_idx", F.lit(0).alias("clause_idx"),
          F.col("Statement").alias("text"), "norm_key", "DetectedRule", "MessageType",
          F.lit("HIGH").alias("Confidence")))

# Clauses from ACTIONS (newly classified)
acts = classified_with_rules.select("correlation_id","sec_ord","stmt_idx","clause_idx",
          F.col("clause_text").alias("text"), F.col("NormalizedStatement").alias("norm_key"),
          "DetectedRule","MessageType","Confidence")

# Combine and mark duplicates
w = Window.partitionBy("correlation_id","MessageType","norm_key") \
          .orderBy("sec_ord","stmt_idx","clause_idx")
audit = whole.select(base).unionByName(acts.select(base)) \
             .withColumn("is_duplicate", F.row_number().over(w) > 1)

# Aggregate into joined messages
def joined(mt):
    items = F.collect_list(F.when((F.col("MessageType") == mt) & ~F.col("is_duplicate"),
                F.struct("sec_ord","stmt_idx","clause_idx","text")))
    return F.concat_ws("\n", F.transform(F.sort_array(items), lambda x: x["text"]))

agg = audit.groupBy("correlation_id").agg(
    joined("CUSTOMER").alias("CustomerMessage"), 
    joined("AGENT").alias("AgentMessage"),
    F.sum((F.col("Confidence") == "LOW").cast("int")).alias("low_conf_count"),
    F.count(F.when(F.col("is_duplicate"), 1)).alias("duplicate_statement_count"))

# ---- Final: join all original columns ----
final = (s2.select("correlation_id", "conversationid", "sessionstarttime", "chattimestamp", "agentemail",
                   "ReasonForCall", "ActionsAndAgreements", "NextSteps", 
                   "parse_status", "has_unknown_header", "embedded_corr_id")
         .join(agg, "correlation_id", "left")
         .fillna({"CustomerMessage": "", "AgentMessage": "", "low_conf_count": 0, 
                  "duplicate_statement_count": 0}))

# Reorder for readability
final_reordered = final.select(
    "correlation_id",

    "low_conf_count",
    "duplicate_statement_count"
)
