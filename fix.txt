def R(rule, msg, conf):
    return F.struct(F.lit(rule).alias("DetectedRule"), F.lit(msg).alias("MessageType"),
                    F.lit(conf).alias("Confidence"))
c = F.col
def add_rules(df):
    dec = (F.when(c("role_to_inf"), R("ROLE_TO_INFINITIVE","AGENT","LOW"))
     .when(c("subject_role") == "NONE",    R("NO_SUBJECT","AGENT","LOW"))
     .when(c("subject_role") == "PRONOUN", R("UNRESOLVED_PRONOUN","AGENT","LOW"))
     .when(c("subject_role") == "BOTH",    R("JOINT_SUBJECT","AGENT","LOW"))
     .when((c("voice") == "PASSIVE") & (c("by_role") == "CUSTOMER"), R("PASSIVE_BY_CUSTOMER","CUSTOMER","HIGH"))
     .when((c("voice") == "PASSIVE") & c("by_role").isin("AGENT","THIRD_PARTY","OTHER"), R("PASSIVE_BY_OTHER","AGENT","HIGH"))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE") & c("state_pred"), R("CUSTOMER_STATE","CUSTOMER","MEDIUM"))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE"), R("CUSTOMER_AS_RECIPIENT","AGENT","HIGH"))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "COPULA") & c("state_pred"), R("CUSTOMER_STATE","CUSTOMER","MEDIUM"))
     .when((c("subject_role") == "CUSTOMER") & (c("voice") == "COPULA"), R("COPULA_UNKNOWN_PREDICATE","AGENT","LOW"))
     .when(c("subject_role") == "CUSTOMER", R("CUSTOMER_AS_ACTOR","CUSTOMER","HIGH"))
     .when(c("subject_role") == "AGENT",    R("AGENT_ACTOR","AGENT","HIGH"))
     .when(c("subject_role") == "THIRD_PARTY", R("THIRD_PARTY_ACTOR","AGENT","HIGH"))
     .otherwise(R("OTHER_ACTOR","AGENT","MEDIUM")))
    return (df.withColumn("dec", dec).select("*", "dec.*").drop("dec")
      .withColumn("Confidence", F.when(c("conj_voice_mismatch") & (c("Confidence") == "HIGH"), "LOW")
                                 .otherwise(c("Confidence")))
      .withColumn("DetectedActor", c("subject_text"))
      .withColumn("IsCustomerSubject", c("subject_role") == "CUSTOMER")
      .withColumn("IsCustomerPassive", (c("subject_role") == "CUSTOMER") & (c("voice") == "PASSIVE"))
      .withColumn("NormalizedStatement", norm_key("clause_text")))
